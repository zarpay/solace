def self.build(
  amount:,
  mint_index:,
  mint_authority_index:,
  destination_index:,
  program_index:
)
  Solace::Instruction.new.tap do |ix|
    ix.program_index = program_index
    ix.accounts = [mint_index, destination_index, mint_authority_index]
    ix.data = data(amount)
  end
end
